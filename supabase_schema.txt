-- 1. Create Advisors Table
CREATE TABLE advisors (
    id TEXT PRIMARY KEY,
    name TEXT NOT NULL,
    username TEXT UNIQUE NOT NULL,
    password TEXT NOT NULL,
    default_zone TEXT DEFAULT 'A',
    schedule TEXT,
    active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT TIMEZONE('utc', NOW())
);

-- 2. Create Students Table
CREATE TABLE students (
    id TEXT PRIMARY KEY,
    name TEXT NOT NULL,
    username TEXT UNIQUE NOT NULL,
    password TEXT NOT NULL,
    program TEXT DEFAULT 'MSc',
    enroll_year INTEGER,
    active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT TIMEZONE('utc', NOW())
);

-- 3. Create Units Table
CREATE TABLE units (
    id INTEGER PRIMARY KEY,
    name TEXT NOT NULL,
    zone TEXT DEFAULT 'A',
    room TEXT,
    zone_idx INTEGER,
    status TEXT DEFAULT 'active',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT TIMEZONE('utc', NOW())
);

-- 4. Create Session Advisors Table
CREATE TABLE session_advisors (
    id TEXT PRIMARY KEY,
    date DATE NOT NULL,
    session TEXT NOT NULL,
    zone_a_id TEXT REFERENCES advisors(id),
    zone_b_id TEXT REFERENCES advisors(id),
    zone_c_id TEXT REFERENCES advisors(id),
    notes TEXT,
    UNIQUE(date, session)
);

-- 5. Create Reservations Table
CREATE TABLE reservations (
    id TEXT PRIMARY KEY,
    student_id TEXT REFERENCES students(id),
    student_name TEXT,
    unit_id INTEGER REFERENCES units(id),
    date DATE NOT NULL,
    session TEXT NOT NULL,
    patient_name TEXT,
    hn TEXT,
    treatment TEXT,
    status TEXT DEFAULT 'confirmed',
    overbooked BOOLEAN DEFAULT FALSE,
    is_ghost BOOLEAN DEFAULT FALSE,
    inherit_unit BOOLEAN DEFAULT FALSE,
    added_by_admin BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT TIMEZONE('utc', NOW()),
    UNIQUE(unit_id, date, session)
);

-- 6. Create Admins Table
CREATE TABLE admins (
    id TEXT PRIMARY KEY,
    name TEXT NOT NULL,
    username TEXT UNIQUE NOT NULL,
    password TEXT NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT TIMEZONE('utc', NOW())
);

-- 7. Create Monthly Lineups Table
CREATE TABLE monthly_lineups (
    month_key TEXT NOT NULL, 
    dow INTEGER NOT NULL, 
    morning_a TEXT,
    morning_b TEXT,
    morning_c TEXT,
    afternoon_a TEXT,
    afternoon_b TEXT,
    afternoon_c TEXT,
    PRIMARY KEY(month_key, dow)
);

-- 8. Create Equipment Table
CREATE TABLE equipment (
    id TEXT PRIMARY KEY,
    category TEXT DEFAULT 'ios',
    name TEXT NOT NULL,
    brand TEXT,
    subtype TEXT,
    serial_number TEXT,
    status TEXT DEFAULT 'active',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT TIMEZONE('utc', NOW())
);

-- 9. Create Equipment Reservations Table
CREATE TABLE equipment_reservations (
    id TEXT PRIMARY KEY,
    student_id TEXT REFERENCES students(id),
    student_name TEXT,
    equipment_id TEXT REFERENCES equipment(id),
    date DATE NOT NULL,
    time_slot TEXT NOT NULL,
    purpose TEXT,
    case_hn TEXT,
    status TEXT DEFAULT 'confirmed',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT TIMEZONE('utc', NOW()),
    UNIQUE(equipment_id, date, time_slot)
);
